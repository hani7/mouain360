<script>
  const hotspots = document.querySelectorAll('.hotspot');
  const img = document.getElementById('floor-plan-img');
  const wrapper = document.getElementById('floor-plan-wrapper');

  hotspots.forEach(hotspot => {
    let offsetX = 0, offsetY = 0;

    hotspot.addEventListener('dragstart', (e) => {
      const rect = wrapper.getBoundingClientRect();
      offsetX = e.clientX - hotspot.getBoundingClientRect().left;
      offsetY = e.clientY - hotspot.getBoundingClientRect().top;
    });

    hotspot.addEventListener('dragend', (e) => {
      const rect = img.getBoundingClientRect();
      const x = ((e.clientX - rect.left - offsetX + 10) / rect.width) * 100;
      const y = ((e.clientY - rect.top - offsetY + 10) / rect.height) * 100;

      hotspot.style.left = `${x}%`;
      hotspot.style.top = `${y}%`;

      const hotspotId = hotspot.dataset.id;
      const csrfToken = '{{ csrf_token }}';

      fetch(`/hotspot/${hotspotId}/update/`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-CSRFToken': csrfToken,
        },
        body: JSON.stringify({ x_percent: x, y_percent: y }),
      }).then(response => {
        if (!response.ok) alert('Failed to update hotspot position');
      });
    });
  });
</script>

<script>
  function copyLink() {
    const dummy = document.createElement("input");
    dummy.value = "{{ request.build_absolute_uri }}";
    document.body.appendChild(dummy);
    dummy.select();
    document.execCommand("copy");
    document.body.removeChild(dummy);
    alert("Link copied to clipboard!");
  }
</script>
{% endblock %}
